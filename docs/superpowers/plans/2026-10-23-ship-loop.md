# Ship Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** After you approve an issue list, every unblocked issue starts in its own T3 thread automatically, blocked issues start when their blockers close, and the epic gets a roll-up retro when all sub-issues close.

**Architecture:** `plan-issues` files issues and runs `bin/ship-loop` once; launchd runs it every 5 minutes. `ship-loop` (bash) reads GitHub labels and `Blocked by #N` lines and calls `bin/t3-run` (Node) for each issue that is ready. `t3-run` talks to the local T3 server over its WebSocket RPC to start one thread in a new worktree on the model for the issue's tier. GitHub labels are the only state.

**Tech Stack:** bash, `gh` 2.102, `jq`, Node 24 (built-in `WebSocket`, no npm deps), launchd, T3 Code server 0.1.2 at `127.0.0.1:3773`.

**Spec:** `docs/superpowers/specs/2026-10-23-ship-loop-design.md`

## Global Constraints

- Owner covered: `ne1cc` only. Local clones at `${GITHUB_ROOT:-$HOME/Github}/<owner>/<repo>`.
- Labels (exact names): `epic`, `exec:flash`, `exec:strong`, `shipped`, `ship-failed`, `retro-posted`.
- Dependency syntax (exact): `Blocked by #N`, one per blocker, same repo only.
- Models: `exec:flash` → GLM 5.3 Flash; `exec:strong` and `retro` → Sonnet 5.5.
- Worktree setting for every run: `{type: "worktree", baseRef: "main", startFromOrigin: true}` (shape confirmed from the T3 client bundle).
- Executor prompt (exact): `Implement GitHub issue #N in owner/repo, following Issue-Driven Work.`
- Retro prompt (exact): `Post the roll-up retrospective on epic #E in owner/repo: link each sub-issue PR and its retrospective, note what was kicked back and what each model did well or badly, then close the epic.`
- Scripts must run under macOS `/bin/bash` 3.2 as well as Homebrew bash (no `mapfile`, guard empty arrays).
- No new dependencies. Never commit secrets (the T3 bearer token is never written to disk or logs).
- Outward-facing steps (spike thread, sandbox repo, live runs) ask the user first.
- `t3` CLI path: `${T3_BIN:-$HOME/.t3/bin/t3}` (not on `PATH`).

```js
#!/usr/bin/env node
// t3-run — start one T3 thread (new worktree from origin/main, model chosen by tier) for a prompt.
// Usage: t3-run <repo-path> <exec:flash|exec:strong|retro> <title> <prompt>
"use strict";
const { execFileSync } = require("node:child_process");
const { readFileSync } = require("node:fs");
const { homedir } = require("node:os");
const { join, resolve } = require("node:path");

const T3_BIN = process.env.T3_BIN || join(homedir(), ".t3/bin/t3");
const RUNTIME = join(homedir(), ".t3/userdata/server-runtime.json");
const MODELS = {
  "exec:flash": null, // SPIKE: modelSelection for GLM 5.3 Flash (## Models)
  "exec:strong": null, // SPIKE: modelSelection for Sonnet 5.5 (## Models)
};
MODELS.retro = MODELS["exec:strong"];
const WORKSPACE = { type: "worktree", baseRef: "main", startFromOrigin: true };

function fail(msg, code = 1) {
  process.stderr.write(`t3-run: ${msg}\n`);
  process.exit(code);
}

const [repoArg, tier, title, prompt] = process.argv.slice(2);
if (!prompt || !(tier in MODELS)) {
  fail("usage: t3-run <repo-path> <exec:flash|exec:strong|retro> <title> <prompt>", 2);
}
const repoPath = resolve(repoArg);

// SPIKE: connect(origin, token) → WebSocket per ## Connect
function connect(origin, token) {}
// SPIKE: encode(id, method, payload) → string, decode(data) → {id, ok, value, error} per ## Frames
function encode(id, method, payload) {}
function decode(data) {}

function rpcClient(ws) {
  let next = 1;
  const pending = new Map();
  ws.addEventListener("message", (ev) => {
    const msg = decode(ev.data);
    const p = msg && pending.get(msg.id);
    if (!p) return;
    pending.delete(msg.id);
    msg.ok ? p.resolve(msg.value) : p.reject(new Error(JSON.stringify(msg.error)));
  });
  return (method, payload) =>
    new Promise((res, rej) => {
      const id = String(next++);
      pending.set(id, { resolve: res, reject: rej });
      ws.send(encode(id, method, payload));
      setTimeout(() => pending.has(id) && (pending.delete(id), rej(new Error(`${method} timed out`))), 30000);
    });
}

/**
 * Starts one thread for the prompt.
 *
 * Contract:
 * - Resolves when T3 accepted the thread start.
 * - Rejects on connect, auth, project, or start errors (caller exits 1).
 *
 * Limits: the session token is revoked by T3 expiry, not here; it is never logged.
 */
async function main() {
  // 1. Server origin from T3's runtime file (port can change between launches)
  const { origin } = JSON.parse(readFileSync(RUNTIME, "utf8"));

  // 2. Scoped bearer token for this one run
  const token = execFileSync(T3_BIN, ["auth", "session", "issue"], { encoding: "utf8" }).trim();

  const ws = await connect(origin, token);
  const call = rpcClient(ws);
  try {
    // 3. Register the repo as a T3 project if it is not one yet
    let projectId = await findProject(call, repoPath);
    if (!projectId) {
      execFileSync(T3_BIN, ["project", "add", repoPath], { stdio: "ignore" });
      projectId = await findProject(call, repoPath);
    }
    if (!projectId) throw new Error(`no T3 project for ${repoPath}`);

    // 4. Start the thread (route from ## Start route)
    await startThread(call, { projectId, title, prompt, modelSelection: MODELS[tier], workspaceStrategy: WORKSPACE });
  } finally {
    ws.close();
  }
}

// SPIKE: findProject(call, path) → projectId | undefined, using the ## Projects method and fields
async function findProject(call, path) {}
// SPIKE: startThread(call, args) per ## Start route (direct create, or upsert → runNow → delete)
async function startThread(call, args) {}

main().catch((err) => fail(err.message));
```

- [ ] **Step 3: Write `bin/ship-loop`**

```bash
#!/usr/bin/env bash
# ship-loop — one pass: start a T3 thread for every unblocked sub-issue of each open ne1cc epic,
#             and one roll-up retro run when all of an epic's sub-issues are closed.
#
# Usage: ship-loop [--repo owner/repo] [--dry-run]
# State lives only in GitHub labels: shipped, ship-failed, retro-posted.
set -euo pipefail

REPO_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
OWNER="${SHIP_OWNER:-ne1cc}"
T3_URL="${T3_URL:-http://127.0.0.1:3773}"
T3_RUN="${T3_RUN:-$REPO_DIR/bin/t3-run}"
LOCK_DIR="${SHIP_LOCK_DIR:-${TMPDIR:-/tmp}/ship-loop.lock}"
GITHUB_ROOT="${GITHUB_ROOT:-$HOME/Github}"
DRY_RUN=false
REPOS=()

while [[ $# -gt 0 ]]; do
  case "$1" in
    --repo) REPOS+=("$2"); shift 2 ;;
    --dry-run) DRY_RUN=true; shift ;;
    -h|--help) echo "Usage: ship-loop [--repo owner/repo] [--dry-run]"; exit 0 ;;
    *) echo "ship-loop: unknown argument: $1" >&2; exit 2 ;;
  esac
done

log() { printf '%s ship-loop: %s\n' "$(date '+%F %T')" "$*"; }
notify() {
  log "FAIL: $*"
  osascript -e 'on run argv' -e 'display notification (item 1 of argv) with title "ship-loop"' -e 'end run' "$*" >/dev/null 2>&1 || true
}
label() { # label <repo> <issue> <add|remove> <name>
  if "$DRY_RUN"; then log "dry-run: $3 $4 on $1#$2"; return 0; fi
  gh issue edit "$2" -R "$1" "--$3-label" "$4" >/dev/null
}
run() { # run <repo> <tier> <title> <prompt>
  if "$DRY_RUN"; then log "dry-run: t3-run $1 $2 \"$3\""; return 0; fi
  "$T3_RUN" "$GITHUB_ROOT/$1" "$2" "$3" "$4"
}

# 1. Lock: plan-issues and launchd can start a pass at the same moment; a crashed pass's lock expires after 30 min
if [[ -d "$LOCK_DIR" ]] && (( $(date +%s) - $(stat -f %m "$LOCK_DIR") > 1800 )); then rmdir "$LOCK_DIR"; fi
mkdir "$LOCK_DIR" 2>/dev/null || { log "another pass is running"; exit 0; }
trap 'rmdir "$LOCK_DIR"' EXIT

# 2. T3 down → nothing can start; skip quietly (any HTTP answer counts as up, so no -f)
curl -s -o /dev/null --max-time 3 "$T3_URL" || { log "T3 not reachable at $T3_URL, skipping"; exit 0; }

# 3. Repos with an open epic, unless --repo was given
if [[ ${#REPOS[@]} -eq 0 ]]; then
  while IFS= read -r r; do [[ -n "$r" ]] && REPOS+=("$r"); done < <(
    gh search issues --owner "$OWNER" --label epic --state open --json repository \
      --jq '.[].repository.nameWithOwner' | sort -u)
fi

for repo in ${REPOS[@]+"${REPOS[@]}"}; do
  # 4. Threads need a local clone to branch worktrees from
  if [[ ! -d "$GITHUB_ROOT/$repo/.git" ]] && ! "$DRY_RUN"; then
    GITHUB_ROOT="$GITHUB_ROOT" "$REPO_DIR/bin/gh-ns-clone" "$repo" >/dev/null || { notify "clone failed: $repo"; continue; }
  fi

  while IFS=$'\t' read -r epic retro_done; do
    subs="$(gh api "repos/$repo/issues/$epic/sub_issues" --paginate --jq '.[]' | jq -s .)"
    [[ "$(jq length <<<"$subs")" -gt 0 ]] || continue

    # 5. Open sub-issues not yet dispatched: start each whose blockers are all closed
    #    (tier "-" = no exec:* label; tab is IFS whitespace, so an empty field would shift columns)
    while IFS=$'\t' read -r n tier title body64; do
      body="$(printf '%s' "$body64" | base64 -d)"
      blocked=""
      for b in $(grep -oE 'Blocked by #[0-9]+' <<<"$body" | grep -oE '[0-9]+$' || true); do
        [[ "$(gh issue view "$b" -R "$repo" --json state --jq .state)" == CLOSED ]] || blocked+=" #$b"
      done
      if [[ -n "$blocked" ]]; then log "$repo#$n waits on$blocked"; continue; fi
      if [[ "$tier" == - ]]; then
        label "$repo" "$n" add ship-failed; notify "$repo#$n has no exec:* label"; continue
      fi
      label "$repo" "$n" add shipped
      if run "$repo" "$tier" "#$n $title" "Implement GitHub issue #$n in $repo, following Issue-Driven Work."; then
        log "started $repo#$n ($tier)"
      else
        label "$repo" "$n" remove shipped; label "$repo" "$n" add ship-failed; notify "could not start $repo#$n"
      fi
    done < <(jq -r '.[]
      | select(.state == "open")
      | select(any(.labels[]; .name == "shipped" or .name == "ship-failed") | not)
      | [.number, ([.labels[].name | select(startswith("exec:"))][0] // "-"), .title, ((.body // "") | @base64)]
      | @tsv' <<<"$subs")

    # 6. Every sub-issue closed → one roll-up retro run; retro-posted stops repeats
    if [[ "$retro_done" == false ]] && jq -e 'all(.[]; .state == "closed")' <<<"$subs" >/dev/null; then
      label "$repo" "$epic" add retro-posted
      if run "$repo" retro "#$epic Roll-up retrospective" "Post the roll-up retrospective on epic #$epic in $repo: link each sub-issue PR and its retrospective, note what was kicked back and what each model did well or badly, then close the epic."; then
        log "started retro for $repo#$epic"
      else
        label "$repo" "$epic" remove retro-posted; label "$repo" "$epic" add ship-failed; notify "could not start retro for $repo#$epic"
      fi
    fi
  done < <(gh issue list -R "$repo" --label epic --state open --json number,labels \
    --jq '.[] | "\(.number)\t\(any(.labels[]; .name == "retro-posted"))"')
done
```
